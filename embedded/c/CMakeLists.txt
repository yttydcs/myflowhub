cmake_minimum_required(VERSION 3.16)
project(myflowhub_embedded_c VERSION 1.0.0 LANGUAGES C)

add_library(myflowhub_embedded STATIC src/mfh_codec.c src/mfh_client.c)
target_include_directories(myflowhub_embedded PUBLIC include)
target_compile_features(myflowhub_embedded PUBLIC c_std_99)
if(MSVC)
  target_compile_options(myflowhub_embedded PRIVATE /W4 /WX)
else()
  target_compile_options(myflowhub_embedded PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(mfh_embedded_tests tests/test_main.c)
  target_link_libraries(mfh_embedded_tests PRIVATE myflowhub_embedded)
  target_compile_definitions(mfh_embedded_tests PRIVATE MFH_FIXTURE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures")
  if(NOT MSVC)
    target_compile_options(mfh_embedded_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME mfh_embedded_tests COMMAND mfh_embedded_tests)
endif()
